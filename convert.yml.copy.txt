name: Convert SC1 to SC2

on:
  workflow_dispatch:
  push:

jobs:
  convert:
    runs-on: windows-2022
    steps:
      - uses: actions/checkout@v4

      - name: Enable long paths
        shell: cmd
        run: git config --system core.longpaths true

      - uses: ilammy/msvc-dev-cmd@v1

      - name: Make sure Ninja exists
        shell: pwsh
        run: |
          if (-not (Get-Command ninja -ErrorAction SilentlyContinue)) { choco install ninja -y --no-progress }
          ninja --version

      - name: Configure
        shell: cmd
        run: cmake -S . -B build -G Ninja -DCMAKE_BUILD_TYPE=Release

      - name: Build flatc
        shell: cmd
        run: cmake --build build --target flatc

      - name: Generate FlatBuffers headers
        shell: pwsh
        run: |
          $flatc = "build/_deps/flatbuffers-build/flatc.exe"
          $jobs = @(
            @{ src = "build/_deps/supercell-texture-src/supercell-texture/sctx_schemas"; out = "build/_deps/supercell-texture-src/supercell-texture/source/texture/SCTX" },
            @{ src = "build/_deps/supercellflash-src/supercell-flash/sc2_schemas"; out = "build/_deps/supercellflash-src/supercell-flash/source/flash/SC2" }
          )
          foreach ($j in $jobs) {
            foreach ($f in Get-ChildItem "$($j.src)/*.fbs") {
              & $flatc --cpp --cpp-std=c++17 -o $j.out $f.FullName
              if ($LASTEXITCODE -ne 0) { throw "flatc failed on $($f.Name)" }
            }
          }

      - name: Build converter
        shell: cmd
        run: cmake --build build

      - name: Convert (raw textures)
        shell: cmd
        run: build\sc1tosc2.exe input -o output

      - name: Convert (KTX textures, optional variant)
        shell: cmd
        continue-on-error: true
        run: build\sc1tosc2.exe input -o output_ktx --ktx

      - uses: actions/upload-artifact@v4
        with:
          name: sc2-result
          path: |
            output
            output_ktx
          if-no-files-found: warn
